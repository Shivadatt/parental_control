// Authentication and authorization helpers for Edge Functions.
// User identity always comes from a verified JWT; device identity always comes
// from x-device-id + x-device-secret with a constant-time SHA-256 comparison.
import {
  createClient,
  type SupabaseClient,
  type User,
} from "npm:@supabase/supabase-js@2";
import { ApiError, ensureNoError } from "./shared-envelope.ts";
import {
  constantTimeEqualHex,
  isUuid,
  sha256Hex,
} from "./shared-validation.ts";

export type Role = "unset" | "parent" | "child";

export interface ProfileRow {
  id: string;
  role: Role;
  display_name: string;
}

export interface UserIdentity {
  user: User;
  emailVerified: boolean;
  accessToken: string;
}

export interface AuthContext extends UserIdentity {
  profile: ProfileRow;
}

export interface DeviceRow {
  id: string;
  family_id: string;
  child_user_id: string;
  secret_hash: string | null;
  status: "pending" | "active" | "revoked";
  credential_revoked_at: string | null;
  display_name: string;
  policy_version: number;
  last_seen_at: string | null;
  battery_level: number | null;
  is_charging: boolean | null;
  enforcement_mode: "none" | "device_admin" | "device_owner";
}

let cachedClient: SupabaseClient | null = null;

/** Service-role client. Called only from Edge Functions server-side. */
export function getServiceClient(): SupabaseClient {
  if (cachedClient) return cachedClient;
  const url = Deno.env.get("SUPABASE_URL");
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
  if (!url || !serviceKey) {
    throw new ApiError(
      "server_misconfigured",
      "The backend is missing required server configuration.",
      503,
    );
  }
  cachedClient = createClient(url, serviceKey, {
    auth: { persistSession: false, autoRefreshToken: false },
    global: { headers: { "X-Client-Info": "parentcontrol-edge/1" } },
  });
  return cachedClient;
}

function bearerToken(req: Request): string {
  const header = req.headers.get("authorization") ?? "";
  const match = /^Bearer\s+(\S+)$/i.exec(header);
  if (!match || match[1].length < 20) {
    throw new ApiError("unauthorized", "Sign in to continue.", 401);
  }
  return match[1];
}

/**
 * Validates the caller JWT with Supabase Auth without requiring an application
 * profile row. Used exclusively by lifecycles (such as account deletion retries)
 * that must proceed even if the user profile has already been removed.
 */
export async function requireAuthUser(
  req: Request,
  db: SupabaseClient,
): Promise<UserIdentity> {
  const token = bearerToken(req);
  let user: User | null = null;
  try {
    const { data, error } = await db.auth.getUser(token);
    if (error) {
      throw new ApiError("unauthorized", "Your session is not valid. Sign in again.", 401);
    }
    user = data.user;
  } catch (error) {
    if (error instanceof ApiError) throw error;
    throw new ApiError("unauthorized", "Your session is not valid. Sign in again.", 401);
  }
  if (!user) {
    throw new ApiError("unauthorized", "Your session is not valid. Sign in again.", 401);
  }
  const emailVerified = Boolean(user.email_confirmed_at);
  return { user, emailVerified, accessToken: token };
}

/**
 * Validates the caller JWT with Supabase Auth, loads the profile and (when
 * requested) requires a confirmed email address.
 */
export async function requireUser(
  req: Request,
  db: SupabaseClient,
  options: { requireVerifiedEmail?: boolean } = {},
): Promise<AuthContext> {
  const { user, emailVerified, accessToken } = await requireAuthUser(req, db);

  const profileResult = await db
    .from("profiles")
    .select("id, role, display_name")
    .eq("id", user.id)
    .maybeSingle();
  ensureNoError(profileResult);
  const row = profileResult.data as ProfileRow | null;
  if (!row) {
    throw new ApiError(
      "profile_missing",
      "The account profile is not ready yet. Try again shortly.",
      403,
    );
  }
  if (row.role !== "unset" && row.role !== "parent" && row.role !== "child") {
    throw new ApiError("profile_invalid", "The account profile is invalid.", 403);
  }
  if (options.requireVerifiedEmail && !emailVerified) {
    throw new ApiError(
      "email_not_verified",
      "Confirm your email address before continuing.",
      403,
    );
  }
  return { user, profile: row, emailVerified, accessToken };
}

export function requireRole(profile: ProfileRow, role: "parent" | "child"): void {
  if (profile.role !== role) {
    throw new ApiError(
      "role_required",
      role === "parent"
        ? "A parent account is required for this action."
        : "A child account is required for this action.",
      403,
    );
  }
}

async function loadFamily(
  db: SupabaseClient,
  familyId: string,
): Promise<{ id: string; deleted_at: string | null }> {
  const { data, error } = await db
    .from("families")
    .select("id, deleted_at")
    .eq("id", familyId)
    .maybeSingle();
  ensureNoError({ error });
  if (!data || (data as { deleted_at: string | null }).deleted_at) {
    throw new ApiError("forbidden", "You do not have access to this family.", 403);
  }
  return data as { id: string; deleted_at: string | null };
}

    .eq("user_id", userId)
    .eq("status", "active")

}

export async function assertFamilyOwner(
  db: SupabaseClient,
  userId: string,
  familyId: string,
): Promise<{ id: string; owner_id: string; settings: Record<string, unknown> }> {
  const { data, error } = await db
    .from("families")
    .select("id, owner_id, settings, deleted_at")
    .eq("id", familyId)
    .maybeSingle();
  ensureNoError({ error });
  const family = data as

    | null;
  if (!family || family.deleted_at) {
    throw new ApiError("forbidden", "You do not have access to this family.", 403);

  await assertFamilyParent(db, userId, familyId);
  if (family.owner_id !== userId) {
    throw new ApiError(

      403,

  }
  return family;
}

/** Loads an active device and verifies that the caller is a parent in its family. */
export async function requireActiveDeviceForParent(
  db: SupabaseClient,
  userId: string,
  deviceId: string,
): Promise<DeviceRow> {
  const { data, error } = await db
    .from("devices")
    .select(
      "id, family_id, child_user_id, secret_hash, status, credential_revoked_at, display_name, policy_version, last_seen_at, battery_level, is_charging, enforcement_mode",
    )
    .eq("id", deviceId)
    .maybeSingle();
  ensureNoError({ error });
  const device = data as DeviceRow | null;
  if (!device) {
    throw new ApiError("device_not_found", "The device was not found.", 404);
  }
  await assertFamilyParent(db, userId, device.family_id);
  if (device.status !== "active") {
    throw new ApiError("device_not_active", "The device is not active.", 403);
  }
  return device;
}

/**
 * Device proof: x-device-id + x-device-secret. A user JWT or device id alone is
 * never accepted.
 */
export async function requireDevice(
  req: Request,
  db: SupabaseClient,
): Promise<DeviceRow> {
  const deviceId = req.headers.get("x-device-id") ?? "";
  const secret = req.headers.get("x-device-secret") ?? "";
  if (!deviceId || !secret) {
    throw new ApiError(
      "device_auth_required",
      "Device credentials are required.",
      401,
    );
  }
  if (!isUuid(deviceId) || secret.length < 16 || secret.length > 512) {
    throw new ApiError(
      "invalid_device_credentials",
      "Device credentials are not valid.",
      401,
    );
  }

  const { data, error } = await db
    .from("devices")
    .select(
      "id, family_id, child_user_id, secret_hash, status, credential_revoked_at, display_name, policy_version, last_seen_at, battery_level, is_charging, enforcement_mode",
    )
    .eq("id", deviceId)
    .maybeSingle();
  ensureNoError({ error });
  const device = data as DeviceRow | null;

  const digest = await sha256Hex(secret);
  if (!device || !device.secret_hash) {
    // Still run a comparison so unknown ids and wrong secrets are similar.
    constantTimeEqualHex(digest, "0".repeat(64));
    throw new ApiError(
      "invalid_device_credentials",
      "Device credentials are not valid.",
      401,
    );
  }
  if (!constantTimeEqualHex(digest, device.secret_hash)) {
    throw new ApiError(
      "invalid_device_credentials",
      "Device credentials are not valid.",
      401,
    );
  }
  if (device.status !== "active" || device.credential_revoked_at !== null) {
    throw new ApiError(
      "device_revoked",
      "This device credential was revoked. Link the device again.",
      403,
    );
  }
  return device;
}

// Pure validation / crypto helpers. No network or Supabase imports so these
// can be unit-tested with `deno test` without credentials.
import { ApiError } from "./shared-envelope.ts";

export const UUID_RE =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
export const HEX64_RE = /^[0-9a-f]{64}$/;
// Android application ids: at least two dot-separated segments.
export const PACKAGE_RE = /^[A-Za-z][A-Za-z0-9_]*(\.[A-Za-z][A-Za-z0-9_]*)+$/;
export const INSTALLATION_ID_RE = /^[A-Za-z0-9._:-]+$/;
// The contract uses this 30-character unambiguous subset.
export const PAIRING_ALPHABET = "23456789ABCDEFGHJKMNPQRSTVWXYZ";
export const PAIRING_CODE_LENGTH = 8;
export const PAIRING_CODE_RE = /^[23456789ABCDEFGHJKMNPQRSTVWXYZ]{8}$/;

// A permanent Child ID is an identifier, never a credential. The "PC" prefix
// makes it impossible to confuse with an 8-character linking code, so the
// server can route on format alone without one path becoming an oracle for the
// other.
export const CHILD_CONTACT_PREFIX = "PC";
export const CHILD_CONTACT_BODY_LENGTH = 8;
export const CHILD_CONTACT_RE = /^PC[23456789ABCDEFGHJKMNPQRSTVWXYZ]{8}$/;

const encoder = new TextEncoder();

export function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

export function requireRecord(
  value: unknown,
  field: string,
): Record<string, unknown> {
  if (!isRecord(value)) {
    throw new ApiError("validation_error", `${field} must be an object.`, 400);
  }
  return value;
}

export interface StringOptions {
  min?: number;
  max?: number;
  trim?: boolean;
  pattern?: RegExp;
  patternMessage?: string;
}

export function readString(
  value: unknown,
  field: string,
  options: StringOptions = {},
): string {
  if (typeof value !== "string") {
    throw new ApiError("validation_error", `${field} must be a string.`, 400);
  }
  let result = options.trim === false ? value : value.trim();
  const min = options.min ?? 0;
  const max = options.max ?? 2000;
  if (result.length < min || result.length > max) {
    throw new ApiError(
      "validation_error",
      `${field} must contain between ${min} and ${max} characters.`,
      400,
    );
  }
  if (options.pattern && !options.pattern.test(result)) {
    throw new ApiError(
      "validation_error",
      options.patternMessage ?? `${field} has an invalid format.`,
      400,
    );
  }
  return result;
}

export function readOptionalString(
  value: unknown,
  field: string,
  options: StringOptions = {},
): string | undefined {
  if (value === undefined || value === null) return undefined;
  if (typeof value === "string" && value.trim().length === 0) return undefined;
  return readString(value, field, options);
}

export function readBoolean(value: unknown, field: string): boolean {

export function readOptionalBoolean(

export function normalizePairingCode(raw: string): string {

/** True for a canonical (unformatted) Child ID such as `PCAB12CD34`. */
export function isChildContactId(value: unknown): value is string {
  return typeof value === "string" && CHILD_CONTACT_RE.test(value);

  return isPairingCode(value) || isChildContactId(value);

/** `PCAB12CD34` → `PC-AB12CD34` for display. Canonical form is what is stored. */

  return `${canonical.slice(0, CHILD_CONTACT_PREFIX.length)}-${
    canonical.slice(CHILD_CONTACT_PREFIX.length)

function randomAlphabetString(length: number): string {
  const alphabetLength = PAIRING_ALPHABET.length; // 30
  const limit = Math.floor(256 / alphabetLength) * alphabetLength; // 240
  const bytes = new Uint8Array(16);
  let result = "";
  while (result.length < length) {
    crypto.getRandomValues(bytes);
    for (const byte of bytes) {
      if (byte < limit && result.length < length) {
        result += PAIRING_ALPHABET[byte % alphabetLength];
      }
    }
  }
  return result;
}

/** Cryptographically random 8-character code from the contract alphabet. */
export function generatePairingCode(): string {
  return randomAlphabetString(PAIRING_CODE_LENGTH);
}

/** Cryptographically random canonical Child ID, e.g. `PC7K4M2QRX`. */
export function generateChildContactId(): string {
  return CHILD_CONTACT_PREFIX + randomAlphabetString(CHILD_CONTACT_BODY_LENGTH);
}

export async function sha256Hex(value: string): Promise<string> {
  const digest = await crypto.subtle.digest("SHA-256", encoder.encode(value));
  return Array.from(new Uint8Array(digest))
    .map((byte) => byte.toString(16).padStart(2, "0"))
    .join("");
}

// Emitted once per isolate so a missing pepper is visible without flooding logs.
let pairingHashPepperWarned = false;

/**
 * Keyed digest used for stored pairing-attempt lookups.
 *
 * The identifier space (30^8) is small enough that an unsalted SHA-256 can be
 * brute-forced from a database dump, so pairing_attempts must use an HMAC keyed
 * by a server-side secret. When no pepper is configured we fall back to the
 * legacy plain SHA-256 so the function remains deployable without the secret.
 */
export async function identifierHash(
  value: string,
  pepper?: string,
): Promise<string> {
  const key = pepper ?? Deno.env.get("PAIRING_HASH_PEPPER") ?? "";
  if (key.length === 0) {
    if (!pairingHashPepperWarned) {
      pairingHashPepperWarned = true;
      console.warn("pairing_hash_pepper_missing");
    }
    return await sha256Hex(value);
  }
  const cryptoKey = await crypto.subtle.importKey(
    "raw",
    encoder.encode(key),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );
  const signature = await crypto.subtle.sign(
    "HMAC",
    cryptoKey,
    encoder.encode(value),
  );
  return Array.from(new Uint8Array(signature))
    .map((byte) => byte.toString(16).padStart(2, "0"))
    .join("");
}

export function constantTimeEqual(a: string, b: string): boolean {
  const left = encoder.encode(a);
  const right = encoder.encode(b);
  const length = Math.max(left.length, right.length);
  let diff = left.length ^ right.length;
  for (let i = 0; i < length; i += 1) {
    diff |= (left[i] ?? 0) ^ (right[i] ?? 0);
  }
  return diff === 0;
}

export function constantTimeEqualHex(a: string, b: string): boolean {
  return constantTimeEqual(a.toLowerCase(), b.toLowerCase());
}

/** Compares shared secrets without leaking length through early exit. */
export async function constantTimeSecretEqual(
  provided: string,
  expected: string,
): Promise<boolean> {
  const [left, right] = await Promise.all([sha256Hex(provided), sha256Hex(expected)]);
  return constantTimeEqualHex(left, right);
}

/** Validates a YYYY-MM-DD calendar date and returns it unchanged. */
export function readDateOnly(value: unknown, field: string): string {
  if (typeof value !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    throw new ApiError("validation_error", `${field} must be YYYY-MM-DD.`, 400);
  }
  const [year, month, day] = value.split("-").map(Number);
  const parsed = new Date(Date.UTC(year, month - 1, day));
  if (
    Number.isNaN(parsed.getTime()) ||
    parsed.getUTCFullYear() !== year ||
    parsed.getUTCMonth() !== month - 1 ||
    parsed.getUTCDate() !== day
  ) {
    throw new ApiError("validation_error", `${field} is not a real date.`, 400);
  }
  return value;
}

/** Haversine distance in metres. */
export function distanceMeters(
  lat1: number,
  lng1: number,
  lat2: number,
  lng2: number,
): number {
  const earthRadius = 6371000;
  const toRad = (degrees: number) => (degrees * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2;
  return 2 * earthRadius * Math.asin(Math.min(1, Math.sqrt(a)));
}

export function dedupeStrings(values: string[]): string[] {
  return Array.from(new Set(values));
}
